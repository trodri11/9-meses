        function drawBracelet() {
            const canvas = document.getElementById('bracelet-canvas');
            if (!canvas) return;
            const ctx = canvas.getContext('2d');
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            const textInput = document.getElementById('bracelet-text').value.toUpperCase().trim() || "LOVER";
            const characters = textInput.split('');

            // Canvas Dimensions & Center
            const centerX = canvas.width / 2;
            const centerY = canvas.height / 2;
            const radiusX = 220;
            const radiusY = 110;

            // Draw String/Thread behind
            ctx.beginPath();
            ctx.ellipse(centerX, centerY, radiusX, radiusY, 0, 0, 2 * Math.PI);
            ctx.strokeStyle = '#94a3b8';
            ctx.lineWidth = 3;
            ctx.stroke();

            // Total beads (Text characters + filler beads)
            const totalBeads = Math.max(32, characters.length + 18);
            const textStartIndex = Math.floor((totalBeads - characters.length) / 2);

            for (let i = 0; i < totalBeads; i++) {
                const angle = (i / totalBeads) * 2 * Math.PI - Math.PI / 2;
                const x = centerX + radiusX * Math.cos(angle);
                const y = centerY + radiusY * Math.sin(angle);

                const charIndex = i - textStartIndex;
                const isTextBead = charIndex >= 0 && charIndex < characters.length;

                if (isTextBead) {
                    // Letter Bead (Round white with letter)
                    ctx.beginPath();
                    ctx.arc(x, y, 14, 0, 2 * Math.PI);
                    ctx.fillStyle = '#ffffff';
                    ctx.shadowColor = 'rgba(0, 0, 0, 0.4)';
                    ctx.shadowBlur = 6;
                    ctx.fill();
                    ctx.strokeStyle = '#cbd5e1';
                    ctx.lineWidth = 1.5;
                    ctx.stroke();
                    ctx.shadowBlur = 0;

                    // Draw Letter
                    ctx.fillStyle = currentPalette.letters;
                    ctx.font = 'bold 13px "Plus Jakarta Sans", sans-serif';
                    ctx.textAlign = 'center';
                    ctx.textBaseline = 'middle';
                    ctx.fillText(characters[charIndex], x, y + 1);
                } else {
                    // Colored Spacer / Decorative Bead
                    const colorIndex = i % currentPalette.beads.length;
                    const beadColor = currentPalette.beads[colorIndex];

                    ctx.beginPath();
                    ctx.arc(x, y, 11, 0, 2 * Math.PI);
                    ctx.fillStyle = beadColor;
                    ctx.shadowColor = 'rgba(0, 0, 0, 0.3)';
                    ctx.shadowBlur = 4;
                    ctx.fill();
                    ctx.shadowBlur = 0;

                    // Shiny Highlight on bead
                    ctx.beginPath();
                    ctx.arc(x - 3, y - 3, 3, 0, 2 * Math.PI);
                    ctx.fillStyle = 'rgba(255, 255, 255, 0.5)';
                    ctx.fill();
                }
            }
        }

        function searchLyrics() {
            const query = document.getElementById('lyrics-search-input').value.toLowerCase().trim();
            const resultsContainer = document.getElementById('lyrics-results');
            resultsContainer.innerHTML = '';

            if (!query) {
                resultsContainer.innerHTML = `
                    <div class="col-span-full text-center py-12 text-slate-400 glass-card rounded-2xl">
                        <i class="fa-solid fa-compact-disc text-4xl mb-3 text-cyan-400 animate-spin" style="animation-duration: 8s;"></i>
                        <p class="text-base">Digite qualquer termo acima (como <strong>'romeu'</strong>, <strong>'scarf'</strong>, <strong>'midnight'</strong> ou <strong>'lover'</strong>) para buscar!</p>
                    </div>
                `;
                return;
            }

            // Normalizing text for Portuguese query matching (e.g. romeu -> romeo / love story)
            let searchTerms = [query];
            if (query === 'romeu') searchTerms.push('romeo');

            const filteredSongs = taylorSongsDatabase.filter(song => {
                const lyricsLower = song.lyrics.toLowerCase();
                const titleLower = song.title.toLowerCase();
                return searchTerms.some(term => lyricsLower.includes(term) || titleLower.includes(term));
            });

            if (filteredSongs.length === 0) {
                resultsContainer.innerHTML = `
                    <div class="col-span-full text-center py-12 text-slate-400 glass-card rounded-2xl">
                        <i class="fa-solid fa-face-sad-tear text-3xl mb-2 text-pink-400"></i>
                        <p>Nenhuma música encontrada com a palavra "<strong>${query}</strong>". Tente outro trecho!</p>
                    </div>
                `;
                return;
            }

            filteredSongs.forEach(song => {
                // Highlight search word in lyrics text
                let highlightedLyrics = song.lyrics;
                searchTerms.forEach(term => {
                    const regex = new RegExp(`(${term})`, 'gi');
                    highlightedLyrics = highlightedLyrics.replace(regex, `<mark class="bg-pink-500/40 text-pink-200 px-1 rounded">$1</mark>`);
                });

                const cardHtml = `
